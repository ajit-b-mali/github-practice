now I am doing deploy.yml

```

name: Deploy to VPS

on:
  push:
    branches:
      - main
  workflow_dispatch:

jobs:
  deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Configure SSH
        run: |
          mkdir -p ~/.ssh
          chmod 700 ~/.ssh

          printf '%s\n' "${{ secrets.VPS_SSH_KEY }}" > ~/.ssh/deploy_key
          printf '%s\n' "${{ secrets.VPS_KNOWN_HOSTS }}" > ~/.ssh/known_hosts

          chmod 600 ~/.ssh/deploy_key
          chmod 600 ~/.ssh/known_hosts

      - name: Deploy to VPS
        env:
          VPS_HOST: ${{ secrets.VPS_HOST }}
          VPS_PORT: ${{ secrets.VPS_PORT }}
          VPS_USER: ${{ secrets.VPS_USER }}
        run: |
          ssh -i ~/.ssh/deploy_key \
            -p "$VPS_PORT" \
            -o IdentitiesOnly=yes \
            -o StrictHostKeyChecking=yes \
            "$VPS_USER@$VPS_HOST" \
            'cd /opt/apps/github-practice && git pull --ff-only origin main && sudo /usr/bin/systemctl restart github-practice && sudo /usr/bin/systemctl is-active github-practice'
```

this workflow does following things

1. runs whenever I push to main
2. prepares ssh credentials on GitHub runner
3. connects to vps as deploy
4. pulls the latest code from GitHub
5. restarts Node.js systemd service
6. checks that the service is active
